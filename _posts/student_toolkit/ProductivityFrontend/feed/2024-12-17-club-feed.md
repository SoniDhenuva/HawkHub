<script type="module">
import { fetchFeed, fetchSavedPosts, createFeedPost, likePost, savePost, deletePost }
  from '{{ site.baseurl }}/assets/js/api/clubFeed.js';

// ── Jekyll-injected data ─────────────────────────────────────────────────
const SITE_BASE = '{{ site.baseurl }}';
const ALL_CLUBS = {{ site.data.school_clubs | jsonify }};

// ── State ────────────────────────────────────────────────────────────────
let posts         = [];
let currentType   = '';
let savedMode     = false;
let searchTimer;
const imageQueue  = [];   // { dataUrl }
const MAX_IMAGES  = 5;

// ── Bootstrap ────────────────────────────────────────────────────────────
async function init() {
  buildClubCheckboxes();
  bindToolbar();
  bindSidebar();
  bindModal();
  bindForm();
  if (new URLSearchParams(location.search).get('create') === '1') openModal();
  await loadFeed();
}

// ── Club checkboxes ──────────────────────────────────────────────────────
function buildClubCheckboxes() {
  document.getElementById('clubCheckboxes').innerHTML = ALL_CLUBS.map(c => `
    <label class="club-check-label">
      <input type="checkbox" name="club" value="${esc(c.id)}">
      <span class="club-check-pill">${esc(c.name)}</span>
    </label>`).join('');
}

// ── Load ─────────────────────────────────────────────────────────────────
async function loadFeed() {
  document.getElementById('instaFeed').innerHTML =
    `<div class="feed-loading"><div class="spinner"></div><p>Loading…</p></div>`;
  const search = document.getElementById('feedSearch').value.trim();
  try {
    posts = savedMode
      ? (await fetchSavedPosts() || [])
      : (await fetchFeed({ postType: currentType, search }) || []);
    setStatus('');
  } catch (err) {
    console.warn('Feed error:', err);
    posts = [];
    setStatus('Could not reach the server.');
  }
  renderFeed();
}

// ── Render ───────────────────────────────────────────────────────────────
function renderFeed() {
  if (!posts.length) {
    document.getElementById('instaFeed').innerHTML =
      `<div class="feed-empty"><p>No posts here yet.</p></div>`;
    return;
  }
  document.getElementById('instaFeed').innerHTML = posts.map(buildCard).join('');
  bindCards();
}

function buildCard(post) {
  const club      = ALL_CLUBS.find(c => c.id === post.clubId) || null;
  const clubName  = club ? club.name : (post.clubId || 'Unknown Club');
  const logoSrc   = club?.image ? `${SITE_BASE}/images/${esc(club.image)}` : null;
  const images    = post.imageUrls?.length ? post.imageUrls
                    : post.imageUrl ? [post.imageUrl] : [];
  const postType  = post.postType || 'General';
  const caption   = post.caption || post.description || '';
  const tags      = post.tags || [];
  const likeCount = post.likeCount ?? post.likes ?? 0;
  const saveCount = post.saveCount ?? 0;
  const liked     = !!post.likedByCurrentUser;
  const saved     = !!post.savedByCurrentUser;
  const isOwner   = !!post.owner;

  const logoHtml = logoSrc
    ? `<img class="club-logo" src="${esc(logoSrc)}" alt="${esc(clubName)}" loading="lazy">`
    : `<div class="club-logo-placeholder">${esc(clubName.slice(0,2).toUpperCase())}</div>`;

  const deleteBtn = isOwner
    ? `<button class="card-delete-btn" data-id="${esc(post.id)}" title="Delete post" aria-label="Delete post">&#x2715;</button>`
    : '';

  const carouselHtml = buildCarousel(images, post.id);

  const tagsHtml = tags.length
    ? `<div class="card-tags">${tags.map(t => `<span class="tag-chip">${esc(t)}</span>`).join('')}</div>`
    : '';

  return `
    <article class="insta-card" data-id="${esc(post.id)}">
      <div class="card-header">
        ${logoHtml}
        <div class="card-header-info">
          <span class="card-club-name">${esc(clubName)}</span>
          <div class="card-meta">
            <span>${relativeTime(post.createdAt)}</span>
            <span class="card-type-badge">${esc(postType)}</span>
          </div>
        </div>
        ${deleteBtn}
      </div>

      ${carouselHtml}

      <div class="card-body">
        <p class="card-caption">${escHtml(caption)}</p>
        ${tagsHtml}
      </div>

      <div class="card-actions">
        <button class="action-btn like-btn${liked ? ' liked' : ''}" data-id="${esc(post.id)}">
          ${heartSvg(liked)}<span class="like-count">${likeCount}</span>
        </button>
        <button class="action-btn save-btn${saved ? ' saved' : ''}" data-id="${esc(post.id)}">
          ${bookmarkSvg(saved)}<span class="save-count">${saveCount}</span>
        </button>
      </div>
    </article>`;
}

function buildCarousel(images, postId) {
  if (!images.length) return '';
  if (images.length === 1) return `
    <div class="carousel-wrap">
      <div class="carousel-track">
        <div class="carousel-slide"><img src="${esc(images[0])}" alt="Post image" loading="lazy"></div>
      </div>
    </div>`;

  const slides = images.map(src =>
    `<div class="carousel-slide"><img src="${esc(src)}" alt="Post image" loading="lazy"></div>`
  ).join('');
  const dots = images.map((_, i) =>
    `<span class="carousel-dot${i === 0 ? ' active' : ''}" data-index="${i}"></span>`
  ).join('');

  return `
    <div class="carousel-wrap" data-carousel="${esc(postId)}">
      <div class="carousel-track">${slides}</div>
      <button class="carousel-btn prev" aria-label="Previous">&#8249;</button>
      <button class="carousel-btn next" aria-label="Next">&#8250;</button>
      <div class="carousel-counter"><span class="cur-slide">1</span> / ${images.length}</div>
      <div class="carousel-dots">${dots}</div>
    </div>`;
}

// ── Card event listeners ─────────────────────────────────────────────────
function bindCards() {
  // Carousels
  document.querySelectorAll('.carousel-wrap[data-carousel]').forEach(wrap => {
    const track  = wrap.querySelector('.carousel-track');
    const slides = wrap.querySelectorAll('.carousel-slide');
    const dots   = wrap.querySelectorAll('.carousel-dot');
    const counter = wrap.querySelector('.cur-slide');
    let cur = 0;

    const go = idx => {
      cur = Math.max(0, Math.min(idx, slides.length - 1));
      track.style.transform = `translateX(-${cur * 100}%)`;
      dots.forEach((d, i) => d.classList.toggle('active', i === cur));
      wrap.querySelector('.prev').disabled = cur === 0;
      wrap.querySelector('.next').disabled = cur === slides.length - 1;
      if (counter) counter.textContent = cur + 1;
    };
    go(0);
    wrap.querySelector('.prev').addEventListener('click', () => go(cur - 1));
    wrap.querySelector('.next').addEventListener('click', () => go(cur + 1));
    dots.forEach((d, i) => d.addEventListener('click', () => go(i)));
  });

  // Likes
  document.querySelectorAll('.like-btn').forEach(btn => {
    btn.addEventListener('click', async () => {
      btn.disabled = true;
      const id = btn.dataset.id;
      try {
        const res   = await likePost(id);
        const isOn  = res?.liked ?? !btn.classList.contains('liked');
        const count = res?.likeCount ?? (parseInt(btn.querySelector('.like-count').textContent) + (isOn ? 1 : -1));
        btn.classList.toggle('liked', isOn);
        btn.innerHTML = `${heartSvg(isOn)}<span class="like-count">${count}</span>`;
        const p = posts.find(p => String(p.id) === String(id));
        if (p) { p.likedByCurrentUser = isOn; p.likeCount = count; }
      } catch { setStatus('Could not like. Are you logged in?'); }
      finally { btn.disabled = false; }
    });
  });

  // Saves
  document.querySelectorAll('.save-btn').forEach(btn => {
    btn.addEventListener('click', async () => {
      btn.disabled = true;
      const id = btn.dataset.id;
      try {
        const res   = await savePost(id);
        const isOn  = res?.saved ?? !btn.classList.contains('saved');
        const count = res?.saveCount ?? (parseInt(btn.querySelector('.save-count').textContent) + (isOn ? 1 : -1));
        btn.classList.toggle('saved', isOn);
        btn.innerHTML = `${bookmarkSvg(isOn)}<span class="save-count">${count}</span>`;
        const p = posts.find(p => String(p.id) === String(id));
        if (p) { p.savedByCurrentUser = isOn; p.saveCount = count; }
      } catch { setStatus('Could not save. Are you logged in?'); }
      finally { btn.disabled = false; }
    });
  });

  // Delete (owner only)
  document.querySelectorAll('.card-delete-btn').forEach(btn => {
    btn.addEventListener('click', async () => {
      if (!confirm('Delete this post?')) return;
      btn.disabled = true;
      const id = btn.dataset.id;
      try {
        await deletePost(id);
        posts = posts.filter(p => String(p.id) !== String(id));
        document.querySelector(`.insta-card[data-id="${id}"]`)?.remove();
        if (!posts.length) renderFeed();
      } catch { setStatus('Could not delete post.'); btn.disabled = false; }
    });
  });

  // Tag click → search
  document.querySelectorAll('.tag-chip').forEach(chip => {
    chip.addEventListener('click', () => {
      document.getElementById('feedSearch').value = chip.textContent;
      currentType = '';
      savedMode   = false;
      document.querySelectorAll('.filter-btn').forEach(b => b.classList.toggle('active', b.dataset.type === ''));
      document.getElementById('savedTabBtn').classList.remove('is-active');
      loadFeed();
    });
  });
}

// ── Image upload ──────────────────────────────────────────────────────────
function bindForm() {
  const zone  = document.getElementById('uploadZone');
  const input = document.getElementById('fileInput');

  zone.addEventListener('click',    () => imageQueue.length < MAX_IMAGES && input.click());
  zone.addEventListener('dragover', e => { e.preventDefault(); zone.classList.add('drag'); });
  zone.addEventListener('dragleave',  () => zone.classList.remove('drag'));
  zone.addEventListener('drop', e => {
    e.preventDefault(); zone.classList.remove('drag');
    [...e.dataTransfer.files].forEach(addImage);
  });
  input.addEventListener('change', e => { [...e.target.files].forEach(addImage); input.value = ''; });
  document.getElementById('createForm').addEventListener('submit', handleSubmit);
}

function addImage(file) {
  if (!file.type.startsWith('image/')) return;
  if (imageQueue.length >= MAX_IMAGES) {
    setStatus(`Maximum ${MAX_IMAGES} images per post.`);
    return;
  }
  const reader = new FileReader();
  reader.onload = ev => { imageQueue.push({ dataUrl: ev.target.result }); renderPreviews(); };
  reader.readAsDataURL(file);
}

function renderPreviews() {
  const wrap = document.getElementById('uploadPreviews');
  wrap.innerHTML = imageQueue.map((img, i) => `
    <div class="upload-thumb">
      <img src="${img.dataUrl}" alt="">
      <span class="upload-order-badge">${i + 1}</span>
      <button class="upload-thumb-remove" data-idx="${i}" type="button">&times;</button>
    </div>`).join('');
  wrap.querySelectorAll('.upload-thumb-remove').forEach(btn =>
    btn.addEventListener('click', () => {
      imageQueue.splice(Number(btn.dataset.idx), 1);
      renderPreviews();
    })
  );
  document.getElementById('uploadLimitNote').textContent =
    `${imageQueue.length} / ${MAX_IMAGES} images`;
}

// ── Submit ────────────────────────────────────────────────────────────────
async function handleSubmit(e) {
  e.preventDefault();
  const clubs   = [...document.querySelectorAll('#clubCheckboxes input:checked')].map(i => i.value);
  const caption = document.getElementById('captionInput').value.trim();
  if (!clubs.length) { setStatus('Select at least one club.'); return; }
  if (!caption)      { setStatus('Caption is required.');      return; }

  const btn = document.getElementById('submitBtn');
  btn.disabled = true; btn.textContent = 'Posting…';
  try {
    await createFeedPost({
      caption,
      clubIds:  clubs,
      postType: document.getElementById('postType').value,
      images:   imageQueue.map(i => i.dataUrl),
    });
    resetForm(); closeModal(); setStatus('');
    await loadFeed();
  } catch (err) {
    setStatus('Failed to post. Are you logged in? ' + (err.message || ''));
  } finally {
    btn.disabled = false; btn.textContent = 'Post to Feed';
  }
}

function resetForm() {
  document.getElementById('createForm').reset();
  imageQueue.length = 0;
  renderPreviews();
  document.querySelectorAll('#clubCheckboxes input').forEach(i => i.checked = false);
}

function relativeTime(iso) {
  if (!iso) return '';
  const d = new Date(iso); if (isNaN(d)) return '';
  const ms = Date.now() - d, m = Math.floor(ms / 60000);
  if (m < 1)  return 'Just now';
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const dy = Math.floor(h / 24);
  return dy < 7 ? `${dy}d ago` : d.toLocaleDateString();
}
function esc(v) {
  return String(v ?? '').replace(/[&<>"']/g, c =>
    ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[c]));
}
function escHtml(v) { return esc(v); }
