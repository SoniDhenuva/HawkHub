<script type="module">
  import { fetchClubPosts, createClubPost, likeClubPost } from '{{ site.baseurl }}/assets/js/api/clubFeed.js';

  const LOCAL_POSTS_KEY = 'clubFeedDraftPosts';
  let allClubs = [];
  let allPosts = [];
  let currentFilter = 'all';
  let imageData = null;
  let usingLocalFallback = false;

  async function init() {
    loadClubs();
    setupEventListeners();
    await loadPosts();
  }

  function loadClubs() {
    allClubs = {{ site.data.school_clubs | jsonify }};
    renderClubFilters();
    populateClubSelect();
  }

  function renderClubFilters() {
    const filterContainer = document.getElementById('clubFilters');
    filterContainer.innerHTML = allClubs.map(club => `
      <button class="club-filter-btn" data-club="${escapeAttribute(club.id)}">
        ${escapeHTML(club.name)}
      </button>
    `).join('');

    document.querySelectorAll('.club-filter-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('.club-filter-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        currentFilter = btn.dataset.club;
        renderFeed();
      });
    });
  }

  // Populate club select dropdown
  function populateClubSelect() {
    const select = document.getElementById('clubSelect');
    select.innerHTML = '<option value="">-- Choose a club --</option>' + allClubs.map(club => `
      <option value="${escapeAttribute(club.id)}" data-name="${escapeAttribute(club.name)}">${escapeHTML(club.name)}</option>
    `).join('');
  }

  async function loadPosts() {
    try {
      allPosts = await fetchClubPosts({ clubId: currentFilter });
      usingLocalFallback = false;
      setStatus('');
    } catch (error) {
      console.warn('Spring club feed is not available; using local draft posts.', error);
      usingLocalFallback = true;
      allPosts = loadLocalPosts();
      setStatus('Spring is not connected yet, so this page is showing local draft posts on this browser.');
    }
    renderFeed();
  }

  function renderFeed() {
    const grid = document.getElementById('clubFeedGrid');
    
    let filteredPosts = allPosts;
    if (currentFilter !== 'all') {
      filteredPosts = allPosts.filter(post => post.clubId === currentFilter);
    }

    if (filteredPosts.length === 0) {
      grid.innerHTML = `
        <div class="empty-state">
          <p class="empty-state-text">No posts yet. Be the first to create one.</p>
        </div>
      `;
      return;
    }

    grid.innerHTML = filteredPosts.map(post => {
      const club = allClubs.find(c => c.id === post.clubId);
      const postedTime = formatTime(new Date(post.createdAt));
      const imageUrl = post.imageUrl || post.imageData;
      const description = post.description || post.content || '';
      const likes = Number(post.likes || post.likeCount || 0);
      
      return `
        <article class="club-post-card" data-post-id="${escapeAttribute(post.id)}">
          <div class="club-post-image-container">
            ${imageUrl ? `
              <img src="${escapeAttribute(imageUrl)}" class="club-post-image" alt="${escapeAttribute(post.title)}">
            ` : `
              <div class="club-post-image-placeholder">No Image</div>
            `}
            ${club ? `
              <div class="club-badge">
                ${club.image ? `<img src="{{ site.baseurl }}/images/${escapeAttribute(club.image)}" class="club-badge-image" alt="${escapeAttribute(club.name)}">` : ''}
                <span>${escapeHTML(club.name)}</span>
              </div>
            ` : ''}
          </div>
          <div class="club-post-content">
            <h3 class="club-post-title">${escapeHTML(post.title)}</h3>
            <p class="club-post-description">${escapeHTML(description)}</p>
            <div class="club-post-meta">
              <span class="club-post-time">${postedTime}</span>
              <div class="club-post-actions">
                <button class="action-btn like-btn" data-post-id="${escapeAttribute(post.id)}" title="Like post">Like ${likes}</button>
                <button class="action-btn share-btn" data-post-id="${escapeAttribute(post.id)}" title="Copy share link">Share</button>
              </div>
            </div>
          </div>
        </article>
      `;
    }).join('');

    grid.querySelectorAll('.like-btn').forEach(button => {
      button.addEventListener('click', handleLikeClick);
    });

    grid.querySelectorAll('.share-btn').forEach(button => {
      button.addEventListener('click', handleShareClick);
    });
  }

  function formatTime(date) {
    if (Number.isNaN(date.getTime())) return 'Recently';
    const now = new Date();
    const diffMs = now - date;
    const diffMins = Math.floor(diffMs / 60000);
    const diffHours = Math.floor(diffMs / 3600000);
    const diffDays = Math.floor(diffMs / 86400000);

    if (diffMins < 1) return 'Just now';
    if (diffMins < 60) return `${diffMins}m ago`;
    if (diffHours < 24) return `${diffHours}h ago`;
    if (diffDays < 7) return `${diffDays}d ago`;
    
    return date.toLocaleDateString();
  }

  function setupEventListeners() {
    document.getElementById('newPostBtn').addEventListener('click', () => {
      document.getElementById('createPostModal').classList.add('active');
    });

    document.getElementById('closeModal').addEventListener('click', () => {
      document.getElementById('createPostModal').classList.remove('active');
    });

    document.getElementById('createPostModal').addEventListener('click', (e) => {
      if (e.target.id === 'createPostModal') {
        document.getElementById('createPostModal').classList.remove('active');
      }
    });

    const uploadArea = document.getElementById('imageUploadArea');
    const imageInput = document.getElementById('imageInput');

    uploadArea.addEventListener('click', () => imageInput.click());
    
    uploadArea.addEventListener('dragover', (e) => {
      e.preventDefault();
      uploadArea.classList.add('dragging');
    });

    uploadArea.addEventListener('dragleave', () => {
      uploadArea.classList.remove('dragging');
    });

    uploadArea.addEventListener('drop', (e) => {
      e.preventDefault();
      uploadArea.classList.remove('dragging');
      if (e.dataTransfer.files.length) {
        handleImageUpload(e.dataTransfer.files[0]);
      }
    });

    imageInput.addEventListener('change', (e) => {
      if (e.target.files.length) {
        handleImageUpload(e.target.files[0]);
      }
    });

    document.getElementById('createPostForm').addEventListener('submit', handlePostSubmit);
  }

  function handleImageUpload(file) {
    if (!file.type.startsWith('image/')) {
      alert('Please choose an image file.');
      return;
    }

    const reader = new FileReader();
    reader.onload = (e) => {
      imageData = e.target.result;
      document.getElementById('imagePreview').innerHTML = `
        <div class="image-preview">
          <img src="${imageData}" alt="Preview">
        </div>
      `;
    };
    reader.readAsDataURL(file);
  }

  async function handlePostSubmit(e) {
    e.preventDefault();
    
    const clubId = document.getElementById('clubSelect').value;
    const title = document.getElementById('postTitle').value;
    const description = document.getElementById('postDescription').value;
    const submitBtn = document.getElementById('submitBtn');

    if (!clubId || !title || !description) {
      alert('Please fill in all fields');
      return;
    }

    submitBtn.disabled = true;
    submitBtn.textContent = 'Posting...';

    try {
      const postData = {
        clubId,
        title,
        description,
        content: description,
        imageUrl: imageData,
        topicPath: '/student/club-feed',
        createdAt: new Date().toISOString()
      };

      if (usingLocalFallback) {
        saveLocalPost(postData);
      } else {
        await createClubPost(postData);
      }

      document.getElementById('createPostForm').reset();
      document.getElementById('imagePreview').innerHTML = '';
      imageData = null;
      document.getElementById('createPostModal').classList.remove('active');
      await loadPosts();
    } catch (error) {
      console.error('Error creating post:', error);
      alert('Failed to create post. Please try again.');
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Post to Feed';
    }
  }

  async function handleLikeClick(event) {
    const postId = event.currentTarget.dataset.postId;
    event.currentTarget.disabled = true;

    try {
      if (usingLocalFallback) {
        const posts = loadLocalPosts().map(post => {
          if (String(post.id) !== String(postId)) return post;
          return { ...post, likes: Number(post.likes || 0) + 1 };
        });
        saveLocalPosts(posts);
        allPosts = posts;
      } else {
        const updatedPost = await likeClubPost(postId);
        allPosts = allPosts.map(post => String(post.id) === String(postId) ? updatedPost : post);
      }
      renderFeed();
    } catch (error) {
      console.error('Error liking post:', error);
      alert('Could not like this post yet.');
    } finally {
      event.currentTarget.disabled = false;
    }
  }

  async function handleShareClick(event) {
    const postId = event.currentTarget.dataset.postId;
    const shareUrl = `${window.location.origin}{{ site.baseurl }}/student/club-feed?post=${encodeURIComponent(postId)}`;

    try {
      await navigator.clipboard.writeText(shareUrl);
      setStatus('Share link copied.');
      window.setTimeout(() => setStatus(usingLocalFallback ? 'Spring is not connected yet, so this page is showing local draft posts on this browser.' : ''), 2200);
    } catch {
      prompt('Copy this share link:', shareUrl);
    }
  }

  function loadLocalPosts() {
    try {
      return JSON.parse(localStorage.getItem(LOCAL_POSTS_KEY)) || [];
    } catch {
      return [];
    }
  }

  function saveLocalPost(postData) {
    const localPost = {
      ...postData,
      id: `local-${Date.now()}`,
      likes: 0,
    };
    saveLocalPosts([localPost, ...loadLocalPosts()]);
  }

  function saveLocalPosts(posts) {
    localStorage.setItem(LOCAL_POSTS_KEY, JSON.stringify(posts));
  }

  function setStatus(message) {
    const status = document.getElementById('feedStatus');
    status.textContent = message;
    status.classList.toggle('active', Boolean(message));
  }

  function escapeHTML(value) {
    return String(value ?? '').replace(/[&<>"']/g, character => ({
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#039;',
    }[character]));
  }

  function escapeAttribute(value) {
    return escapeHTML(value).replace(/`/g, '&#096;');
  }

  init();
</script>
